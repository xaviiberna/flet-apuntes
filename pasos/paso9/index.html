    # PASO 9a - Primera clase: Tarea (cada tarea es un control personalizado)
# La app sigue siendo funcional (main), pero cada tarea ya es una clase.
from typing import Callable

import flet as ft


@ft.control
class Tarea(ft.Column):
    nombre: str = ""  # campo: texto de la tarea
    # Callbacks: funciones que pasa el "dueño" de la lista.
    # Se llaman on_...: convención de Flet para funciones y eventos
    on_borrar: Callable | None = None  # al pulsar borrar
    on_cambio: Callable | None = None  # al marcar/desmarcar

    def init(self):
        # Se ejecuta al crear la tarea: aquí se construye su interfaz
        self.completada = False
        self.casilla = ft.Checkbox(
            value=False, label=self.nombre, on_change=self.cambio_estado
        )
        self.controls = [
            ft.Row(
                alignment=ft.MainAxisAlignment.SPACE_BETWEEN,
                controls=[
                    self.casilla,
                    ft.IconButton(
                        icon=ft.Icons.DELETE_OUTLINE,
                        tooltip="Eliminar tarea",
                        on_click=self.clic_borrar,
                    ),
                ],
            )
        ]

    def cambio_estado(self, e):
        self.completada = self.casilla.value
        self.on_cambio()

    def clic_borrar(self, e):
        self.on_borrar(self)


def main(page: ft.Page):
    page.title = "Lista de tareas"
    page.padding = 20

    titulo = ft.Text("Mis tareas", size=30, weight=ft.FontWeight.BOLD)
    nueva_tarea = ft.TextField(
        label="Nueva tarea", expand=True, on_submit=lambda e: añadir_tarea(e)
    )
    contador = ft.Text("Tareas pendientes: 0")
    tareas = ft.Column()

    def actualizar_contador():
        pendientes = sum(1 for tarea in tareas.controls if not tarea.completada)
        contador.value = f"Tareas pendientes: {pendientes}"
        page.update()

    def eliminar_tarea(tarea):
        tareas.controls.remove(tarea)
        actualizar_contador()

    def añadir_tarea(e):
        texto = nueva_tarea.value.strip()

        if not texto:
            nueva_tarea.error = "Escribe una tarea"
            page.update()
            return

        nueva_tarea.error = None
        tareas.controls.append(
            Tarea(nombre=texto, on_borrar=eliminar_tarea, on_cambio=actualizar_contador)
        )
        nueva_tarea.value = ""
        actualizar_contador()

    boton = ft.Button(content="Añadir", on_click=añadir_tarea)

    page.add(titulo, ft.Row([nueva_tarea, boton]), contador, tareas)


ft.run(main)
    Líneas Qué hacen
    1-2 Comentarios: qué es este paso.
    3 from typing import Callable: tipo «función», para anotar los callbacks.
    5 Importa Flet.
    8 @ft.control: convierte la clase en un control de Flet.
    9 class Tarea(ft.Column): una tarea es una columna (hereda de Column).
    10 nombre: str = "": campo; se rellena al crear Tarea(nombre="Pan").
    13-14 on_borrar y on_cambio: funciones que le pasa main (callbacks). None = ninguna por defecto.
    16 def init(self): se ejecuta al crear la tarea; aquí se construye lo que se ve.
    18 self.completada = False: dato propio; la tarea empieza pendiente.
    19-21 self.casilla: el Checkbox con el nombre; al marcarlo llama a self.cambio_estado.
    22-34 self.controls: lo que se dibuja. Una Row con la casilla y la papelera; SPACE_BETWEEN las separa a los extremos.
    27-31 IconButton papelera con tooltip (texto al pasar el ratón). on_click=self.clic_borrar: no hace falta lambda, self ya es la tarea.
    36-38 cambio_estado: guarda si está marcada en self.completada y avisa a main con self.on_cambio().
    40-41 clic_borrar: avisa a main pasándole la propia tarea (self).
    44-53 main: página, título, campo (Enter con lambda), contador y columna de tareas, igual que en el paso 8.
    55-58 actualizar_contador: cuenta las tareas con tarea.completada en vez de fila.controls[0].value.
    60-62 eliminar_tarea(tarea): quita la tarea que le pasa clic_borrar y recalcula.
    64-70 añadir_tarea: lee el texto y, si está vacío, muestra error y sale.
    72-75 Crea Tarea(...) pasándole el texto y las dos funciones (callbacks) y la añade a la lista.
    76-77 Vacía el campo y recalcula el contador.
    79-81 Botón «Añadir» y page.add con todo.
    84 Arranca la app.

@ft.control
class AppTareas(ft.Column):
    def init(self):
        self.nueva_tarea = ft.TextField(
            label="Nueva tarea", expand=True, on_submit=self.clic_añadir
        )
        self.tareas = ft.Column()
        self.contador = ft.Text("Tareas pendientes: 0")

        self.total_creadas = 0
        self.texto_total_creadas = ft.Text("Total de tareas creadas: 0")


        self.controls = [
            ft.Text("Mis tareas", size=30, weight=ft.FontWeight.BOLD),
            ft.Row(
                [
                    self.nueva_tarea,
                    ft.Button(content="Añadir", on_click=self.clic_añadir),
                ]
            ),
            self.contador,
            self.texto_total_creadas,
            self.tareas,
        ]

    def clic_añadir(self, e):
        texto = self.nueva_tarea.value.strip()

        if not texto:
            self.nueva_tarea.error = "Escribe una tarea"
            self.update()
            return

        self.nueva_tarea.error = None
        self.tareas.controls.append(
            Tarea(
                nombre=texto,
                on_borrar=self.eliminar_tarea,
                on_cambio=self.actualizar_contador,
            )
        )
        self.nueva_tarea.value = ""
        self.total_creadas += 1
        self.texto_total_creadas.value= f"Total de tareas creadas: {self.total_creadas}"
        self.actualizar_contador()

    def eliminar_tarea(self, tarea):
        self.tareas.controls.remove(tarea)
        self.actualizar_contador()

    def actualizar_contador(self):
        pendientes = sum(1 for tarea in self.tareas.controls if not tarea.completada)
        self.contador.value = f"Tareas pendientes: {pendientes}"
        self.update()



def main(page: ft.Page):
    page.title = "Lista de tareas"
    page.padding = 20
    page.add(AppTareas())


ft.run(main)

class AppTareas(ft.Column):

        )

                ]

            self.contador,

        if not texto:

            Tarea(nombre=texto, on_borrar=self.eliminar_tarea, on_cambio=self.update)
        )
        self.nueva_tarea.value = ""
        self.update()

    def eliminar_tarea(self, tarea):
        self.tareas.controls.remove(tarea)
        self.update()

    def before_update(self):
        # Flet lo llama justo antes de cada self.update()
        pendientes = sum(1 for tarea in self.tareas.controls if not tarea.completada)
        self.contador.value = f"Tareas pendientes: {pendientes}"

    Qué significa cada pieza nueva¶

    Pieza Qué es
    @ft.control Decorador: convierte la clase en un control de Flet (como una dataclass). Los campos con tipo (nombre: str = "") se pueden pasar al crear: Tarea(nombre="Pan").
    class Tarea(ft.Column) La clase hereda de Column: es una columna con tus propias cosas.
    def init(self) Se ejecuta al crear el control: aquí se construye su interfaz (no es __init__).
    on_borrar, on_cambio Callbacks: funciones que pasa el «dueño» de la lista. Prefijo on_ = convención de Flet.
    Callable \| None = None Tipo: «una función o nada».
    self.casilla, self.completada Datos propios de cada tarea: ya no hace falta fila.controls[0].value.
    before_update(self) Flet lo llama justo antes de cada self.update().
    on_cambio=self.update Se pasa el método sin paréntesis como callback.

    Qué se gana

    • t.completada y t.nombre en lugar de fila.controls[0].value / .label (se acabaron los índices mágicos).
    • Cada tarea sabe pintarse y reaccionar a sus eventos.
    • Con before_update, añadir/borrar/marcar solo hacen self.update() y el contador se recalcula solo.

    Ordenar aquí es más fácil

    Con clases la lambda queda más corta: sort(key=lambda t: t.nombre.lower()) → Ordenar en la versión con clases.
